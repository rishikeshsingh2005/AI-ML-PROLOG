% =========================================================
% RULES FOR CLASSIFICATION
% =========================================================

% ---------------------------------------------------------
% RULE 1: TOPPER
% A student is a topper if marks are 75 or above.
% ---------------------------------------------------------

topper(Name) :-
    student(ID, Name),
    marks(ID, Marks),
    Marks >= 75.

% ---------------------------------------------------------
% RULE 2: AVERAGE PERFORMER
% A student is average if marks are 50 to 74.
% ---------------------------------------------------------

average(Name) :-
    student(ID, Name),
    marks(ID, Marks),
    Marks >= 50,
    Marks < 75.

% ---------------------------------------------------------
% RULE 3: FAILED STUDENT
% A student fails if marks are below 50.
% ---------------------------------------------------------

fail(Name) :-
    student(ID, Name),
    marks(ID, Marks),
    Marks < 50.


% GENERAL PERFORMANCE RULE


performance(Name, topper) :-
    student(ID, Name),
    marks(ID, Marks),
    Marks >= 75.

performance(Name, average) :-
    student(ID, Name),
    marks(ID, Marks),
    Marks >= 50,
    Marks < 75.

performance(Name, fail) :-
    student(ID, Name),
    marks(ID, Marks),
    Marks < 50.
